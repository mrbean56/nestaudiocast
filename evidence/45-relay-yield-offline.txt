# Le VRAI Relay (relay/cast_relay.py) avec un sink qui compte chaque paquet, un verrou frais,

----- yield_fake.py -----
"""Offline check of the yield branch: a fake sink records every group packet it would send.

No speaker, no group, no TV, no socket: the sink is an in-memory recorder, the TV a dict.
"""
import json, os, sys, time
sys.path.insert(0, "relay")
import cast_relay as R

OWNER = sys.argv[1]
CALLS = []

class Link:
    def keepalive(self):
        CALLS.append("link.keepalive")
    def close(self):
        CALLS.append("link.close")

class Sink:
    """Every method a group packet would leave through, recorded and neutralised."""
    def __init__(self):
        self.link = Link()
        self.session = None
        self.transport = None
        self.content = None
        self.state = "IDLE"
    def _rec(self, name, *a, **k):
        CALLS.append(name)
    connect = lambda self, *a, **k: self._rec("connect")
    reconnect = lambda self, *a, **k: self._rec("reconnect")
    snapshot = lambda self, *a, **k: (self._rec("snapshot"), {"wall": time.time(), "t": 0.0,
                                                             "state": "PLAYING"})[1]
    prepare = lambda self, *a, **k: (self._rec("prepare"), True)[1]
    start = lambda self, *a, **k: (self._rec("start"), {"playerState": "PLAYING"})[1]
    play_now = lambda self, *a, **k: self._rec("play")
    play = lambda self, *a, **k: self._rec("play")
    pause = lambda self, *a, **k: self._rec("pause")
    seek = lambda self, *a, **k: self._rec("seek")
    set_rate = lambda self, *a, **k: self._rec("set_rate")
    set_volume = lambda self, *a, **k: self._rec("set_volume")
    stop = lambda self, *a, **k: self._rec("stop_app")
    close = lambda self, *a, **k: CALLS.append("sink.close")

class Tv:
    def snapshot(self):
        return {"wall": time.time(), "state": "PLAYING", "t": 100.0,
                "content": "AAAAAAAAAAA", "volume": 0.4, "muted": False}

class Audio:
    inflight = set()
    def url_for(self, vid):
        return None
    def fetch_async(self, vid):
        CALLS.append("yt-dlp")

sink, tv = Sink(), Tv()
log = os.path.join(os.path.dirname(OWNER), "relay-yield-fake.jsonl")
r = R.Relay(tv, sink, Audio(), poll=0.25, log_path=log, verbose=False, owner_path=OWNER,
            mirror_volume=True)

def lock(doc, age=1.0):
    with open(OWNER, "w") as fh:
        json.dump(doc, fh)
    t = time.time() - age
    os.utime(OWNER, (t, t))

lock({"owner": "cast-sender", "pid": 999999, "since": time.time() - 2})
for _ in range(4):
    r.step()
print(json.dumps({"phase": "yielded", "steps": 4, "group_calls": CALLS,
                  "owner_v020": r.owner.v020}))
with open(log) as fh:
    ev = [json.loads(l)["ev"] for l in fh]
print(json.dumps({"phase": "yielded", "log_events": ev}))

# heartbeat: two polls 5.2 s apart must leave two relay_yielded lines, still zero packets
CALLS.clear()
time.sleep(5.2)
r.step()
with open(log) as fh:
    beats = [json.loads(l) for l in fh if json.loads(l)["ev"] == "relay_yielded"]
print(json.dumps({"phase": "heartbeat", "group_calls": CALLS, "relay_yielded_lines": len(beats),
                  "last": beats[-1] if beats else None}))

CALLS.clear()
lock({"owner": "cast-sender", "pid": 999999, "since": time.time() - 40}, age=40)
r.step()
print(json.dumps({"phase": "lock_stale", "group_calls": CALLS, "yielded": r._yielded}))

CALLS.clear()
r.step()
print(json.dumps({"phase": "after_resume", "group_calls": CALLS, "yielded": r._yielded,
                  "need_start": r.need_start}))

## 6. Le crochet d'arrêt décide correctement (lib/emergency.py remplacé par un stub)
----- wrapper_decision.sh -----
#!/bin/sh
# Offline check of tools/relay-stop-post.sh: five locks, one STUB lib/emergency.py, so no
# speaker is reachable from the temporary ROOT and nothing can be muted by this test.
set -eu
REPO=${REPO:-$(pwd)}
W=$(mktemp -d)
mkdir -p "$W/lib" "$W/state" "$W/relay" "$W/.venv/bin"
ln -s "$REPO/relay/cast_relay.py" "$W/relay/cast_relay.py"
for f in castv2.py groupguard.py silence.py asset_server.py; do ln -s "$REPO/lib/$f" "$W/lib/$f"; done
printf '#!/bin/sh\nexec %s/.venv/bin/python "$@"\n' "$REPO" > "$W/.venv/bin/python"
chmod +x "$W/.venv/bin/python"
printf 'print("STUB: emergency.py would have run (no speaker touched)")\n' > "$W/lib/emergency.py"
export CASTRELAY_ROOT="$W" GROUP_OWNER_FILE="$W/state/group.owner"

lock() {  # lock <owner> <age_seconds>
  "$REPO/.venv/bin/python" - "$GROUP_OWNER_FILE" "$1" "$2" <<'PY'
import json, os, sys, time
p, owner, age = sys.argv[1], sys.argv[2], float(sys.argv[3])
t = time.time() - age
json.dump({"owner": owner, "pid": 999999, "since": t}, open(p, "w"))
os.utime(p, (t, t))
PY
}

echo "case 1: fresh cast-sender lock";  lock cast-sender 1;  sh "$REPO/tools/relay-stop-post.sh"; echo "rc=$?"
echo "case 2: stale cast-sender lock";  lock cast-sender 40; sh "$REPO/tools/relay-stop-post.sh"; echo "rc=$?"
echo "case 3: malformed lock";          printf '{"owner": "cast-sender"' > "$GROUP_OWNER_FILE"; sh "$REPO/tools/relay-stop-post.sh"; echo "rc=$?"
echo "case 4: absent lock";            rm -f "$GROUP_OWNER_FILE"; sh "$REPO/tools/relay-stop-post.sh"; echo "rc=$?"
echo "case 5: our own (cast-relay) lock"; lock cast-relay 1; sh "$REPO/tools/relay-stop-post.sh"; echo "rc=$?"
